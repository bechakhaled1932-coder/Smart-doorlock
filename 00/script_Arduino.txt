#include <Keypad.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

LiquidCrystal_I2C lcd(0x27, 16, 2);

const byte ROWS = 4;
const byte COLS = 3;
char keys[ROWS][COLS] = {
  {'1','2','3'},
  {'4','5','6'},
  {'7','8','9'},
  {'*','0','#'}
};
byte rowPins[ROWS] = {8, 7, 6, 5};
byte colPins[COLS] = {4, 3, 2};

Keypad keypad = Keypad(makeKeymap(keys), rowPins, colPins, ROWS, COLS);

// --- Liste des codes et identifiants associés ---
struct Utilisateur {
  String code;
  String nom;
};

Utilisateur utilisateurs[] = {
  {"2003", "Khaled"},
  {"1234", "Ali"},
  {"1111", "Amine"}
};

const int NB_UTILISATEURS = sizeof(utilisateurs) / sizeof(utilisateurs[0]);

String codeSaisi = "";
String message = "";

int relaisPin = 13;
bool relaisActif = false;

int mode = 0;  // 1 = Face, 2 = QR, 3 = Clavier
unsigned long dernierActivation = 0;
const unsigned long timeout = 5000;  // 5 secondes

void setup() {
  Serial.begin(9600);
  lcd.init();
  lcd.backlight();

  pinMode(relaisPin, OUTPUT);
  digitalWrite(relaisPin, LOW);

  afficherMenu();
}

void loop() {
  // Lecture messages série
  while (Serial.available()) {
    char c = Serial.read();
    message += c;
    if (c == '\n') {
      message.trim();
      traiterMessage(message);
      message = "";
    }
  }

  char key = keypad.getKey();

  if (mode == 0 && key) {
    if (key == '1') {
      mode = 1;
      Serial.println("MODE:1");
      lcd.clear();
      lcd.print("Mode: Face ID");
      Serial.println("DETECTION:START");
    } else if (key == '2') {
      mode = 2;
      Serial.println("MODE:2");
      lcd.clear();
      lcd.print("Mode: QR Code");
      Serial.println("DETECTION:START");
    } else if (key == '3') {
      mode = 3;
      Serial.println("MODE:3");
      lcd.clear();
      lcd.print("Mode: Clavier");
      lcd.setCursor(0, 1);
      lcd.print("Saisir le code:");
    }
  }

  if (mode == 3) {
    gererClavier(key);
  }

  // Si relais actif, vérifier timeout
  if (relaisActif && (millis() - dernierActivation >= timeout)) {
    digitalWrite(relaisPin, LOW);
    relaisActif = false;

    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Acces termine");
    delay(2000);

    // Retour à l’état initial
    afficherMenu();
    mode = 0;
    codeSaisi = "";
  }
}

void afficherMenu() {
  lcd.clear();
  lcd.setCursor(0, 0);
  lcd.print("Choix mode:");
  lcd.setCursor(0, 1);
  lcd.print("1:Face 2:QR 3:PIN");
}

void traiterMessage(String msg) {
  if (msg.startsWith("MSG:")) {
    String contenu = msg.substring(4);
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print(contenu.substring(0, 16));
    lcd.setCursor(0, 1);
    if (contenu.length() > 16) {
      lcd.print(contenu.substring(16));
    }
  }
  else if (msg.startsWith("WELCOME:")) {
    String name = msg.substring(8);
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Bienvenue " + name);
    lcd.setCursor(0, 1);
    lcd.print("Acces autorise");
    digitalWrite(relaisPin, HIGH);
    relaisActif = true;
    dernierActivation = millis();
  }
  else if (msg == "ACCESS_DENIED") {
    lcd.clear();
    lcd.setCursor(0, 0);
    lcd.print("Acces refuse");
    lcd.setCursor(0, 1);
    lcd.print("Veuillez reessayer");
    digitalWrite(relaisPin, LOW);
    relaisActif = false;
  }
  else if (msg == "RELAIS_OFF") {
    digitalWrite(relaisPin, LOW);
    relaisActif = false;
  }
}

void gererClavier(char key) {
  if (key) {
    if (key == '#') {
      lcd.clear();
      bool trouve = false;
      for (int i = 0; i < NB_UTILISATEURS; i++) {
        if (codeSaisi == utilisateurs[i].code) {
          lcd.setCursor(0, 0);
          lcd.print("Bienvenue ");
          lcd.print(utilisateurs[i].nom);
          digitalWrite(relaisPin, HIGH);
          relaisActif = true;
          dernierActivation = millis();
          trouve = true;
          break;
        }
      }
      if (!trouve) {
        lcd.setCursor(0, 0);
        lcd.print("Code incorrect");
      }
      delay(2000);
      codeSaisi = "";
      lcd.clear();
      lcd.setCursor(0, 0);
      lcd.print("Saisir le code:");
    }
    else if (key == '*') {
      codeSaisi = "";
      lcd.setCursor(0, 1);
      lcd.print("                ");
      lcd.setCursor(0, 1);
    }
    else {
      if (codeSaisi.length() < 16) {
        codeSaisi += key;
        lcd.print("*");
      }
    }
  }
}
